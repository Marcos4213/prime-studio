function esc(s) {
  return String(s == null ? "" : s).replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));
}

async function api(path, opts = {}) {
  const res = await fetch(path, {
    credentials: "same-origin",
    headers: { "content-type": "application/json" },
    method: opts.method || "GET",
    body: opts.body ? JSON.stringify(opts.body) : undefined,
  });
  let data = {};
  try {
    data = await res.json();
  } catch {
    data = {};
  }
  return { ok: res.ok, status: res.status, data };
}

function spDay(iso) {
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return "";
  return new Intl.DateTimeFormat("en-CA", {
    timeZone: "America/Sao_Paulo",
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
  }).format(d);
}

function formatName(k) {
  if (k === "carousel") return "Carrossel";
  if (k === "video") return "Vídeo";
  if (k === "post") return "Post";
  return k || "";
}

function safeColor(c) {
  const s = String(c || "").trim();
  if (/^#([0-9a-f]{3}|[0-9a-f]{6})$/i.test(s)) return s;
  if (/^[a-zA-Z]{3,20}$/.test(s)) return s;
  return "";
}

async function copyText(text, btn) {
  const value = String(text || "");
  try {
    await navigator.clipboard.writeText(value);
  } catch {
    const ta = document.createElement("textarea");
    ta.value = value;
    ta.setAttribute("readonly", "");
    ta.style.cssText = "position:fixed;inset:0;opacity:0;pointer-events:none";
    document.body.appendChild(ta);
    ta.select();
    document.execCommand("copy");
    ta.remove();
  }
  const prev = btn.textContent;
  btn.textContent = "Copiado";
  setTimeout(() => {
    btn.textContent = prev;
  }, 1400);
}

function buildPiece(piece) {
  const o = piece.output || {};
  const copies = [];
  const copyBtn = (label, text) => {
    const i = copies.length;
    copies.push(String(text || ""));
    return '<button type="button" class="btn-ghost btn-xs" data-copy-i="' + i + '">Copiar ' + esc(label) + "</button>";
  };
  const field = (label, text, live) =>
    '<p class="lbl">' + label + '</p><p class="copy"' + (live ? " data-live-copy" : "") + ">" + (text ? esc(text) : "—") + "</p>";

  let body = "";
  if (piece.format === "video") {
    const sc = o.script || {};
    const rec = o.recording || {};
    const modeLabel = rec.mode === "narracao" ? "Narração sobre a imagem do espaço" : "Na câmera";
    const tips = Array.isArray(rec.tips) ? rec.tips : [];
    const scriptText = [sc.hook, sc.body, sc.cta].filter(Boolean).join("\n\n");
    body =
      field("Gancho", sc.hook, true) +
      field("Desenvolvimento", sc.body) +
      field("Chamada", sc.cta) +
      '<p class="lbl">Como gravar — ' + esc(modeLabel) + "</p>" +
      '<ul class="tips">' + tips.map((t) => "<li>" + esc(t) + "</li>").join("") + "</ul>" +
      '<div class="actions">' + copyBtn("roteiro", scriptText) + "</div>";
  } else if (piece.format === "carousel") {
    const slides = Array.isArray(o.slides) ? o.slides : [];
    body =
      field("Legenda", o.caption, true) +
      '<div class="actions">' + copyBtn("legenda", o.caption) + "</div>" +
      slides
        .map((s, i) => {
          const name = ORDINALS[i] ? ORDINALS[i] + " tela" : "Tela";
          const title = s && s.title ? " — " + esc(s.title) : "";
          return (
            '<article class="slide"><p class="lbl">' + esc(name) + title + "</p>" +
            '<p class="prompt">' + esc(s && s.imagePrompt) + "</p>" +
            '<div class="actions">' + copyBtn("prompt", s && s.imagePrompt) + "</div></article>"
          );
        })
        .join("");
  } else {
    body =
      field("Legenda", o.caption, true) +
      '<div class="actions">' + copyBtn("legenda", o.caption) + "</div>" +
      '<p class="lbl">Prompt de imagem</p><p class="prompt">' + esc(o.imagePrompt) + "</p>" +
      '<div class="actions">' + copyBtn("prompt", o.imagePrompt) + "</div>";
  }

  const warns = Array.isArray(o.warnings) && o.warnings.length
    ? '<div class="warns"><p><strong>Revisar antes de aprovar</strong></p><ul>' +
      o.warnings.map((w) => "<li>" + esc(w) + "</li>").join("") + "</ul></div>"
    : "";
  const methodName = o.copyMethod && o.copyMethod.name ? o.copyMethod.name : "";
  const methodHtml = methodName
    ? '<p class="method" data-copy-method>Método de copy: ' + esc(methodName) + "</p>" +
      '<p class="hint">Dentro do Método PRIME. A gente escreve no tom da clínica, sem promessa e sem número.</p>'
    : "";
  const src = piece.source === "ia" ? "Gerado por IA" : "Exemplo";
  const srcClass = piece.source === "ia" ? "ok" : "wait";
  const approved = piece.status === "aprovado";
  const st = approved ? "Aprovado" : "Rascunho";
  const title = formatName(piece.format) + (piece.theme ? " · " + piece.theme : "");
  const approve = approved
    ? ""
    : '<div class="actions"><button type="button" class="btn" data-approve>Marcar como aprovado</button></div>' +
      '<p class="note">Isso não publica. A peça espera o ok da Prime e da clínica.</p>';
  const html =
    '<div class="piece-main"><article class="sheet"><p class="badges"><span class="badge ' + srcClass + '">' + esc(src) +
    '</span><span class="badge ' + (approved ? "ok" : "") + '">' + esc(st) + "</span></p><h3>" +
    esc(title) + "</h3>" + methodHtml + body + warns + approve + "</article>" + visualHtml() + "</div>" + chatHtml();
  return { html, copies };
}

function pieceCaption(piece) {
  const o = piece.output || {};
  if (piece.format === "video") {
    const sc = o.script || {};
    return [sc.hook, sc.body, sc.cta].filter(Boolean).join(" ");
  }
  return o.caption || "";
}

function mountPiece(el, piece) {
  const built = buildPiece(piece);
  el.classList.add("piece-board");
  el.innerHTML = built.html;
  el.querySelectorAll("[data-copy-i]").forEach((btn) => {
    btn.addEventListener("click", () => copyText(built.copies[Number(btn.getAttribute("data-copy-i"))], btn));
  });
  const visualApi = wireVisual(el, piece);
  wireChat(el, piece, visualApi);
  const approveBtn = el.querySelector("[data-approve]");
  if (!approveBtn) return;
  approveBtn.addEventListener("click", async () => {
    approveBtn.disabled = true;
    const res = await api("/api/pieces/" + encodeURIComponent(piece.id) + "/approve", { method: "POST" });
    if (res.status === 401) return showLogin();
    if (!res.ok || !res.data.ok) {
      approveBtn.disabled = false;
      return;
    }
    piece.status = "aprovado";
    mountPiece(el, piece);
    loadHistory();
  });
}

function wireVisual(el, piece) {
  const visual = el.querySelector("[data-visual]");
  if (!visual) return;
  const stateVis = { kind: "post", ratio: "4:5", origem: "", busy: false };
  const fallback = visual.querySelector("[data-image-fallback]");
  const figure = visual.querySelector("[data-gen-figure]");
  const img = visual.querySelector("[data-gen-img]");
  const down = visual.querySelector("[data-download]");
  const base = visual.querySelector("[data-post-base]");
  const baseImg = visual.querySelector("[data-base-img]");
  const baseCap = visual.querySelector("[data-base-caption]");
  const compose = visual.querySelector("[data-compose]");
  const photoErr = visual.querySelector("[data-photo-err]");
  const chosen = visual.querySelector("[data-chosen]");
  const usePhoto = visual.querySelector("[data-use-photo]");
  const hooks = { image: null, photo: null, useGenerated: null, usePhoto: null };

  function markChosen(text) {
    if (!chosen) return;
    chosen.hidden = false;
    chosen.textContent = text;
  }

  function showGenerated(image) {
    fallback.hidden = true;
    figure.hidden = false;
    img.src = image.url;
    img.alt = "Imagem gerada para a peça";
    down.href = image.url + "?download=1";
    figure.setAttribute("data-image-model", image.model || "");
  }

  function showPhoto(foto, caption) {
    stateVis.origem = foto.id || "";
    base.hidden = false;
    baseImg.src = foto.url;
    baseImg.alt = "Foto da clínica enviada para a peça";
    baseCap.textContent = caption || pieceCaption(piece) || "A clínica, no tom de vocês.";
    if (usePhoto) usePhoto.hidden = false;
    compose.hidden = false;
  }

  function paintPicks() {
    visual.querySelectorAll("[data-kind]").forEach((btn) => {
      const on = btn.getAttribute("data-kind") === stateVis.kind;
      btn.classList.toggle("is-on", on);
      btn.setAttribute("role", "radio");
      btn.setAttribute("aria-checked", on ? "true" : "false");
    });
    visual.querySelectorAll("[data-ratio]").forEach((btn) => {
      const on = btn.getAttribute("data-ratio") === stateVis.ratio;
      btn.classList.toggle("is-on", on);
      btn.setAttribute("role", "radio");
      btn.setAttribute("aria-checked", on ? "true" : "false");
    });
  }
  paintPicks();
  visual.querySelectorAll("[data-kind]").forEach((btn) => {
    btn.addEventListener("click", () => {
      stateVis.kind = btn.getAttribute("data-kind");
      const preset = { post: "4:5", stories: "9:16", capa: "1:1" };
      stateVis.ratio = preset[stateVis.kind] || stateVis.ratio;
      paintPicks();
    });
  });
  visual.querySelectorAll("[data-ratio]").forEach((btn) => {
    btn.addEventListener("click", () => {
      stateVis.ratio = btn.getAttribute("data-ratio");
      paintPicks();
    });
  });

  async function runImage(useOrigem) {
    if (stateVis.busy) return;
    stateVis.busy = true;
    const make = visual.querySelector("[data-make-image]");
    const regen = visual.querySelector("[data-regen]");
    make.disabled = true;
    regen.disabled = true;
    visual.classList.add("is-busy");
    visual.setAttribute("aria-busy", "true");
    fallback.hidden = false;
    fallback.classList.add("is-wait");
    fallback.classList.remove("is-err");
    fallback.textContent = "Preparando a imagem…";
    try {
      const res = await api("/api/estudio/imagem", {
        method: "POST",
        body: {
          kind: stateVis.kind,
          ratio: stateVis.ratio,
          caption: pieceCaption(piece),
          theme: piece.theme || "",
          origem: useOrigem ? stateVis.origem : "",
        },
      });
      if (res.status === 401) return showLogin();
      if (!res.ok || !res.data.ok || !res.data.image) {
        fallback.hidden = false;
        fallback.classList.add("is-err");
        fallback.classList.remove("is-wait");
        fallback.textContent = res.data.motivo === "cota"
          ? "A cota de imagens acabou por agora. Nada foi publicado."
          : "A imagem não veio agora. Nada foi publicado. Você pode tentar de novo.";
        return;
      }
      showGenerated(res.data.image);
      if (hooks.image) hooks.image(res.data.image);
    } finally {
      stateVis.busy = false;
      make.disabled = false;
      regen.disabled = false;
      visual.classList.remove("is-busy");
      visual.removeAttribute("aria-busy");
    }
  }

  visual.querySelector("[data-make-image]").addEventListener("click", () => runImage(false));
  visual.querySelector("[data-regen]").addEventListener("click", () => runImage(false));
  visual.querySelector("[data-use-image]").addEventListener("click", () => {
    if (hooks.useGenerated) hooks.useGenerated();
    const btn = visual.querySelector("[data-use-image]");
    btn.setAttribute("aria-pressed", "true");
    markChosen("Esta imagem fica na peça. Nada vai ao ar sozinho.");
  });
  if (usePhoto) {
    usePhoto.addEventListener("click", () => {
      if (hooks.usePhoto) hooks.usePhoto();
      usePhoto.setAttribute("aria-pressed", "true");
      markChosen("Esta foto fica na peça. Nada vai ao ar sozinho.");
    });
  }
  compose.addEventListener("click", () => runImage(true));

  const photoStatus = visual.querySelector("[data-photo-status]");
  visual.querySelector("[data-photo]").addEventListener("change", async (e) => {
    const file = e.target.files && e.target.files[0];
    e.target.value = "";
    if (!file) return;
    photoErr.hidden = true;
    if (photoStatus) {
      photoStatus.hidden = false;
      photoStatus.textContent = "Enviando a foto…";
    }
    const body = new FormData();
    body.append("foto", file);
    const res = await fetch("/api/estudio/fotos", { method: "POST", credentials: "same-origin", body });
    let data = {};
    try { data = await res.json(); } catch { data = {}; }
    if (photoStatus) photoStatus.hidden = true;
    if (res.status === 401) return showLogin();
    if (!res.ok || !data.ok) {
      photoErr.hidden = false;
      photoErr.textContent = "A foto não entrou. Nada foi apagado.";
      return;
    }
    showPhoto(data.foto, pieceCaption(piece));
    if (hooks.photo) hooks.photo(data.foto);
  });

  return { stateVis, showGenerated, showPhoto, markChosen, hooks };
}

function wireChat(el, piece, visualApi) {
  const chat = el.querySelector("[data-chat]");
  if (!chat || !visualApi) return;
  if (!piece._chat) {
    piece._chat = {
      target: "legenda",
      legenda: { versions: [{ text: pieceCaption(piece), source: "original", warnings: [] }], index: 0, applied: 0 },
      imagem: { versions: [], index: 0, applied: -1 },
      foto: { versions: [], index: 0, applied: -1 },
    };
  }
  const pack = piece._chat;
  const log = chat.querySelector("[data-chat-log]");
  const preview = chat.querySelector("[data-chat-preview]");
  const versions = chat.querySelector("[data-versions]");
  const warns = chat.querySelector("[data-chat-warns]");
  const input = chat.querySelector("[data-chat-input]");
  const err = chat.querySelector("[data-chat-err]");
  const send = chat.querySelector("[data-chat-send]");
  const useBtn = chat.querySelector("[data-chat-use]");
  const toggle = chat.querySelector("[data-chat-toggle]");
  const wide = window.matchMedia("(min-width: 900px)");
  let busy = false;

  function setChatOpen(open) {
    const show = wide.matches || open;
    chat.classList.toggle("is-open", show);
    el.classList.toggle("is-chat-open", !wide.matches && show);
    if (!toggle) return;
    toggle.setAttribute("aria-expanded", show ? "true" : "false");
    toggle.textContent = !wide.matches && show ? "Recolher conversa" : "Abrir conversa";
  }
  setChatOpen(false);
  if (toggle) {
    toggle.addEventListener("click", () => {
      if (wide.matches) return;
      setChatOpen(!chat.classList.contains("is-open"));
    });
  }
  wide.addEventListener("change", () => setChatOpen(wide.matches));

  function bucket() {
    return pack[pack.target];
  }
  function active() {
    const b = bucket();
    return b.versions[b.index] || null;
  }
  function applyToPiece(name, version) {
    if (name === "legenda") {
      const live = el.querySelector("[data-live-copy]");
      if (live) live.textContent = version.text || "—";
      if (piece.format === "video") {
        const script = { ...((piece.output || {}).script || {}), hook: version.text || "", body: "", cta: "" };
        piece.output = { ...(piece.output || {}), script };
      } else {
        piece.output = { ...(piece.output || {}), caption: version.text || "" };
      }
      return;
    }
    if (name === "imagem" && version.url) visualApi.showGenerated(version);
    if (name === "foto" && version.url) visualApi.showPhoto(version, pieceCaption(piece));
  }
  function paint() {
    chat.querySelectorAll("[data-chat-target]").forEach((btn) => {
      const name = btn.getAttribute("data-chat-target");
      const on = name === pack.target;
      btn.hidden = name !== "legenda" && pack[name].versions.length === 0;
      btn.classList.toggle("is-on", on);
      btn.setAttribute("aria-selected", on ? "true" : "false");
    });
    const empty = chat.querySelector("[data-chat-empty]");
    if (empty) empty.hidden = log.children.length > 0;
    versions.replaceChildren();
    const b = bucket();
    b.versions.forEach((v, i) => {
      const li = document.createElement("li");
      const btn = document.createElement("button");
      btn.type = "button";
      btn.className = "btn-ghost btn-xs" + (i === b.applied ? " is-on" : "");
      const mark = i === b.applied ? " · em uso" : i === b.index ? " · vista" : "";
      btn.textContent = "Versão " + (i + 1) + mark;
      btn.setAttribute("aria-pressed", i === b.applied ? "true" : "false");
      btn.addEventListener("click", () => {
        b.index = i;
        paint();
      });
      li.appendChild(btn);
      versions.appendChild(li);
    });
    const v = active();
    if (pack.target === "legenda") preview.textContent = v && v.text ? v.text : "—";
    else if (v && v.url) preview.textContent = v.source === "foto" ? "Foto da clínica, nesta versão." : "Imagem gerada, nesta versão.";
    else preview.textContent = "Gere ou importe a imagem para pedir um retoque.";
    const list = v && Array.isArray(v.warnings) ? v.warnings : [];
    warns.hidden = !list.length;
    warns.textContent = list.join(" ");
    chat.querySelector("[data-chat-undo]").disabled = !b.versions.length || b.index <= 0;
    const inUse = b.versions.length > 0 && b.index === b.applied;
    useBtn.setAttribute("aria-pressed", inUse ? "true" : "false");
    const useImg = el.querySelector("[data-use-image]");
    const useFoto = el.querySelector("[data-use-photo]");
    if (useImg) useImg.setAttribute("aria-pressed", pack.imagem.applied >= 0 && pack.imagem.index === pack.imagem.applied ? "true" : "false");
    if (useFoto) useFoto.setAttribute("aria-pressed", pack.foto.applied >= 0 && pack.foto.index === pack.foto.applied ? "true" : "false");
  }
  function pushVersion(name, version, focus) {
    const b = pack[name];
    b.versions.push(version);
    const index = b.versions.length - 1;
    b.index = index;
    b.applied = index;
    if (focus) {
      pack.target = name;
      applyToPiece(name, version);
    }
    paint();
  }
  function useIndex(index) {
    const b = bucket();
    if (!b.versions[index]) return;
    b.index = index;
    b.applied = index;
    applyToPiece(pack.target, b.versions[index]);
    paint();
  }

  visualApi.hooks.image = (image) => {
    pushVersion("imagem", { id: image.id, url: image.url, source: "ia", model: image.model || "" }, false);
  };
  visualApi.hooks.photo = (foto) => {
    pushVersion("foto", { id: foto.id, url: foto.url, source: "foto" }, false);
  };
  visualApi.hooks.useGenerated = () => {
    const bucketImg = pack.imagem;
    if (!bucketImg.versions.length) return;
    const src = (el.querySelector("[data-gen-img]") || {}).src || "";
    let index = bucketImg.versions.findIndex((v) => v.url && src.indexOf(v.url) !== -1);
    if (index < 0) index = bucketImg.index;
    pack.target = "imagem";
    useIndex(index);
  };
  visualApi.hooks.usePhoto = () => {
    const bucketFoto = pack.foto;
    if (!bucketFoto.versions.length) return;
    pack.target = "foto";
    useIndex(bucketFoto.index);
  };

  chat.querySelectorAll("[data-chat-target]").forEach((btn) => {
    btn.addEventListener("click", () => {
      pack.target = btn.getAttribute("data-chat-target");
      paint();
    });
  });
  chat.querySelector("[data-chat-undo]").addEventListener("click", () => {
    if (bucket().index > 0) useIndex(bucket().index - 1);
  });
  chat.querySelector("[data-chat-use]").addEventListener("click", () => useIndex(bucket().index));
  send.addEventListener("click", async () => {
    const instruction = input.value.trim();
    err.hidden = true;
    if (!instruction || busy) return;
    const name = pack.target;
    const current = active();
    if (name !== "legenda" && (!current || !current.id)) {
      err.hidden = false;
      err.textContent = "Gere ou importe a imagem antes do retoque.";
      return;
    }
    busy = true;
    send.disabled = true;
    send.textContent = "Aplicando…";
    chat.setAttribute("aria-busy", "true");
    const asked = document.createElement("li");
    asked.className = "from-clinic";
    asked.textContent = instruction;
    log.appendChild(asked);
    log.scrollTop = log.scrollHeight;
    const empty = chat.querySelector("[data-chat-empty]");
    if (empty) empty.hidden = true;
    try {
      const res = await api("/api/estudio/chat", {
        method: "POST",
        body: {
          target: name === "legenda" ? "legenda" : "imagem",
          instruction,
          text: name === "legenda" ? (current && current.text) || "" : pieceCaption(piece),
          kind: visualApi.stateVis.kind,
          ratio: visualApi.stateVis.ratio,
          origem: name === "legenda" ? "" : current.id,
        },
      });
      if (res.status === 401) return showLogin();
      if (!res.ok || !res.data.ok || !res.data.version) {
        err.hidden = false;
        err.textContent = res.data.motivo === "sem_chave"
          ? "A conversa precisa da IA. Nada foi alterado."
          : "O ajuste não veio agora. A versão anterior continua.";
        return;
      }
      input.value = "";
      const made = res.data.version;
      if (name === "legenda") {
        pushVersion("legenda", { text: made.text, warnings: made.warnings || [], source: "ia", model: made.model || "" }, true);
      } else {
        pushVersion(name, { id: made.id, url: made.url, source: "ia", model: made.model || "" }, true);
      }
      const reply = document.createElement("li");
      reply.className = "from-ia";
      reply.textContent = "Nova versão pronta.";
      log.appendChild(reply);
      log.scrollTop = log.scrollHeight;
    } finally {
      busy = false;
      send.disabled = false;
      send.textContent = "Aplicar";
      chat.removeAttribute("aria-busy");
    }
  });

  const leg = pack.legenda.versions[pack.legenda.applied];
  if (leg && pack.legenda.applied > 0) applyToPiece("legenda", leg);
  const imgV = pack.imagem.versions[pack.imagem.applied];
  if (imgV && imgV.url) visualApi.showGenerated(imgV);
  const fotoV = pack.foto.versions[pack.foto.applied];
  if (fotoV && fotoV.url) visualApi.showPhoto(fotoV, pieceCaption(piece));
  paint();
}

async function generate() {
  if (state.busy || !state.format) return;
  if (state.format === "video" && state.mode !== "camera" && state.mode !== "narracao") return;
  state.busy = true;
  syncGen();
  qs("#gen").textContent = "Montando a peça…";
  qs("#genErr").hidden = true;
  const typed = qs("#theme").value.trim();
  const theme = typed || boot.today.angle;
  const body = { format: state.format, theme };
  if (state.format === "video") body.recordingMode = state.mode;
  const res = await api("/api/estudio/generate", { method: "POST", body });
  state.busy = false;
  qs("#gen").textContent = "Montar a peça";
  syncGen();
  if (res.status === 401) return showLogin();
  if (!res.ok || !res.data.ok) {
    const err = qs("#genErr");
    err.hidden = false;
    err.textContent = res.data.error === "modo_invalido"
      ? "Escolha câmera ou narração antes de montar o vídeo."
      : res.data.error === "formato_invalido"
        ? "Escolha o formato de novo."
        : "Não foi possível montar a peça. Nada foi publicado.";
    return;
  }
  const box = qs("#result");
  box.hidden = false;
  mountPiece(box, res.data.piece);
  setTodayStatus(true);
  loadHistory();
  box.scrollIntoView({ block: "nearest" });
}

function drawColors() {
  const box = qs("#colorChips");
  box.replaceChildren();
  state.colors.forEach((c, i) => {
    const chip = document.createElement("span");
    chip.className = "chip";
    const sw = document.createElement("span");
    sw.className = "sw";
    const safe = safeColor(c);
    if (safe) sw.style.background = safe;
    const label = document.createElement("span");
    label.textContent = c;
    const rm = document.createElement("button");
    rm.type = "button";
    rm.textContent = "×";
    rm.setAttribute("aria-label", "Remover cor");
    rm.addEventListener("click", () => {
      state.colors.splice(i, 1);
      drawColors();
    });
    chip.append(sw, label, rm);
    box.append(chip);
  });
}

function drawFonts() {
  const box = qs("#fontChips");
  box.replaceChildren();
  state.fonts.forEach((f, i) => {
    const chip = document.createElement("span");
    chip.className = "chip";
    const label = document.createElement("span");
    label.textContent = f;
    const rm = document.createElement("button");
    rm.type = "button";
    rm.textContent = "×";
    rm.setAttribute("aria-label", "Remover fonte");
    rm.addEventListener("click", () => {
      state.fonts.splice(i, 1);
      drawFonts();
    });
    chip.append(label, rm);
    box.append(chip);
  });
}

async function loadBrand() {
  const res = await api("/api/profile");
  if (res.status === 401) return showLogin();
  const p = res.ok && res.data.ok ? res.data.profile : {};
  state.colors = Array.isArray(p.colors) ? p.colors.slice() : [];
  state.fonts = Array.isArray(p.fonts) ? p.fonts.slice() : [];
  qs("#tone").value = p.tone || "";
  qs("#avoid").value = p.avoid || "";
  qs("#highlights").value = p.highlights || "";
  drawColors();
  drawFonts();
}

async function loadHistory() {
  const list = qs("#histList");
  const res = await api("/api/history");
  if (res.status === 401) return showLogin();
  if (!res.ok || !res.data.ok) {
    list.textContent = "Não foi possível abrir o acervo.";
    return;
  }
  const items = res.data.items || [];
  setTodayStatus(items.some((it) => spDay(it.created_at) === boot.todayKey));
  list.replaceChildren();
  if (!items.length) {
    const p = document.createElement("p");
    p.className = "empty";
    p.textContent = "Nada no acervo ainda. A primeira peça fica guardada aqui, e amanhã você continua daí.";
    list.append(p);
    return;
  }
  items.forEach((it) => {
    const btn = document.createElement("button");
    btn.type = "button";
    btn.className = "hitem";
    const title = document.createElement("span");
    title.className = "hitem-title";
    title.textContent = formatName(it.format) + (it.theme ? " · " + it.theme : "");
    const meta = document.createElement("span");
    meta.className = "hitem-meta";
    const when = new Date(it.created_at).toLocaleString("pt-BR", {
      timeZone: "America/Sao_Paulo",
      dateStyle: "short",
      timeStyle: "short",
    });
    const st = it.status === "aprovado" ? "Aprovado" : "Rascunho";
    const src = it.source === "ia" ? "IA" : "exemplo";
    meta.textContent = when + " · " + st + " · " + src;
    btn.append(title, meta);
    btn.addEventListener("click", () => openPiece(it.id));
    list.append(btn);
  });
}

async function openPiece(id) {
  const detail = qs("#histPiece");
  const close = qs("#histClose");
  detail.textContent = "Abrindo…";
  close.hidden = false;
  const res = await api("/api/pieces/" + encodeURIComponent(id));
  if (res.status === 401) return showLogin();
  if (!res.ok || !res.data.ok) {
    detail.textContent = "Não foi possível abrir esta peça.";
    return;
  }
  mountPiece(detail, res.data.piece);
  detail.scrollIntoView({ block: "nearest" });
}

function applyTheme(theme) {
  if (theme === "dark") document.documentElement.dataset.theme = "dark";
  else delete document.documentElement.dataset.theme;
  document.querySelectorAll("[data-theme-toggle]").forEach((btn) => {
    btn.textContent = theme === "dark" ? "Claro" : "Escuro";
  });
}

function wire() {
  let saved = "";
  try {
    saved = localStorage.getItem("estudio-theme") || "";
  } catch {
    saved = "";
  }
  applyTheme(saved === "dark" ? "dark" : "light");
  document.querySelectorAll("[data-theme-toggle]").forEach((btn) => {
    btn.addEventListener("click", () => {
      const next = document.documentElement.dataset.theme === "dark" ? "light" : "dark";
      applyTheme(next);
      try {
        localStorage.setItem("estudio-theme", next);
      } catch {
        /* preferência fica só nesta visita */
      }
    });
  });

  qs("#loginForm").addEventListener("submit", async (e) => {
    e.preventDefault();
    const err = qs("#loginErr");
    const btn = qs("#loginBtn");
    err.hidden = true;
    btn.disabled = true;
    btn.textContent = "Entrando…";
    const res = await api("/api/login", {
      method: "POST",
      body: { email: qs("#loginEmail").value, password: qs("#loginPassword").value },
    });
    btn.disabled = false;
    btn.textContent = "Entrar";
    if (res.ok && res.data.ok) start();
    else {
      err.hidden = false;
      err.textContent = res.data.error === "credenciais" ? "E-mail ou senha não conferem." : "Não foi possível entrar.";
    }
  });

  qs("#logout").addEventListener("click", async () => {
    await api("/api/logout", { method: "POST" });
    location.reload();
  });

  qs("#useAngle").addEventListener("click", () => {
    qs("#theme").value = boot.today.angle;
    if (state.format) qs("#theme").focus();
  });

  document.querySelectorAll(".fmt").forEach((btn) => {
    btn.addEventListener("click", () => {
      state.format = btn.dataset.format;
      document.querySelectorAll(".fmt").forEach((b) => {
        const on = b === btn;
        b.setAttribute("aria-checked", on ? "true" : "false");
        b.classList.toggle("is-on", on);
      });
      const composer = qs("#composer");
      composer.hidden = false;
      qs("#delivers").textContent = btn.dataset.delivers || "";
      const methodByFormat = { post: "AIDA", carousel: "AIDA", video: "AIDA" };
      const formatMethod = qs("#formatMethod");
      if (formatMethod) {
        formatMethod.hidden = false;
        formatMethod.textContent = "Método de copy: " + (methodByFormat[state.format] || "AIDA");
      }
      const isVideo = state.format === "video";
      qs("#modeBox").hidden = !isVideo;
      if (!isVideo) {
        state.mode = "";
        document.querySelectorAll(".mode").forEach((m) => {
          m.setAttribute("aria-checked", "false");
          m.classList.remove("is-on");
        });
      }
      syncGen();
      composer.scrollIntoView({ block: "nearest" });
    });
  });

  document.querySelectorAll(".mode").forEach((btn) => {
    btn.addEventListener("click", () => {
      state.mode = btn.dataset.mode;
      document.querySelectorAll(".mode").forEach((m) => {
        const on = m === btn;
        m.setAttribute("aria-checked", on ? "true" : "false");
        m.classList.toggle("is-on", on);
      });
      syncGen();
    });
  });

  qs("#gen").addEventListener("click", generate);

  const addColor = () => {
    const val = qs("#colorIn").value.trim();
    if (!val || state.colors.length >= 8) return;
    state.colors.push(val.slice(0, 24));
    qs("#colorIn").value = "";
    drawColors();
  };
  const addFont = () => {
    const val = qs("#fontIn").value.trim();
    if (!val || state.fonts.length >= 6) return;
    state.fonts.push(val.slice(0, 60));
    qs("#fontIn").value = "";
    drawFonts();
  };
  qs("#addColor").addEventListener("click", addColor);
  qs("#addFont").addEventListener("click", addFont);
  qs("#colorIn").addEventListener("keydown", (e) => {
    if (e.key === "Enter") {
      e.preventDefault();
      addColor();
    }
  });
  qs("#fontIn").addEventListener("keydown", (e) => {
    if (e.key === "Enter") {
      e.preventDefault();
      addFont();
    }
  });

  qs("#saveBrand").addEventListener("click", async () => {
    const btn = qs("#saveBrand");
    const msg = qs("#brandMsg");
    btn.disabled = true;
    const res = await api("/api/profile", {
      method: "PUT",
      body: {
        colors: state.colors,
        fonts: state.fonts,
        tone: qs("#tone").value,
        avoid: qs("#avoid").value,
        highlights: qs("#highlights").value,
      },
    });
    btn.disabled = false;
    if (res.status === 401) return showLogin();
    msg.hidden = false;
    msg.textContent = res.ok ? "Marca guardada. A próxima peça já sai neste tom." : "Não foi possível guardar a marca.";
  });

  qs("#histClose").addEventListener("click", () => {
    qs("#histPiece").replaceChildren();
    qs("#histClose").hidden = true;
  });
}

wire();
start();
